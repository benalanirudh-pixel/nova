def normalize_key(
    value: Any,
) -> str:

    text = clean_text(
        value
    ).lower()

    output = []

    for character in text:

        if character.isalnum():

            output.append(
                character
            )

        else:

            output.append("_")

    return (
        "".join(output)
        .strip("_")
    )


def find_column(
    dataframe: pd.DataFrame,
    *candidates: str,
) -> str | None:

    if (
        dataframe is None
        or dataframe.empty
    ):

        return None

    normalized = {
        normalize_key(column):
            column
        for column
        in dataframe.columns
    }

    for candidate in candidates:

        key = normalize_key(
            candidate
        )

        if key in normalized:

            return normalized[key]

    return None
