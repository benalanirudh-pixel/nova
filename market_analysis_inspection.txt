def get_market_analysis() -> str:
    """
    Run the NSE market scanner and return an evidence-grounded
    market-intelligence snapshot.

    READ-ONLY.

    This tool does NOT:
    - place trades
    - connect to a broker
    - generate orders
    - prove institutional activity
    - provide derivatives positioning
    - provide sector analysis unless explicitly available
    - provide India VIX unless explicitly available

    The tool deliberately separates:

    [VERIFIED DATA]
    [CALCULATED DATA]
    [INFERENCE BOUNDARY]
    [NOT AVAILABLE]

    This prevents the LLM from treating missing information
    as if it were measured.
    """

        # ====================================================================
        # RUN EXISTING SCANNER
        # ====================================================================

        dataframe, summary = scan_market()

        if dataframe is None or dataframe.empty:

            return (
                "MARKET ANALYSIS ERROR | "
                "The NSE scanner returned no usable stock data."
            )

        # ====================================================================
        # SUMMARY
        # ====================================================================

        market_regime = summary.get(
            "market_regime",
            "UNKNOWN",
        )

        nifty_returns = summary.get(
            "nifty_returns",
            {},
        )

        breadth = summary.get(
            "breadth",
            {},
        )

        stocks_in_universe = summary.get(
            "stocks_in_universe",
            0,
        )

        stocks_analyzed = summary.get(
            "stocks_analyzed",
            len(dataframe),
        )

        timestamp = summary.get(
            "timestamp",
            "UNKNOWN",
        )

        # ====================================================================
        # OUTPUT
        # ====================================================================

        output = []

        output.append(
            "============================================================"
        )

        output.append(
            "NOVA EVIDENCE-GROUNDED NSE MARKET ANALYSIS"
        )

        output.append(
            "============================================================"
        )

        output.append(
            "ANALYSIS_STATUS=SUCCESS"
        )

        output.append(
            "DATA_TYPE=READ_ONLY_QUANTITATIVE_MARKET_ANALYSIS"
        )

        output.append(
            f"DATA_TIMESTAMP={timestamp}"
        )

        output.append(
            f"STOCKS_IN_UNIVERSE={stocks_in_universe}"
        )

        output.append(
            f"STOCKS_ANALYZED={stocks_analyzed}"
        )

        # ====================================================================
        # EVIDENCE CONTRACT
        # ====================================================================

        output.append("")
        output.append(
            "==================== EVIDENCE CONTRACT ===================="
        )

        output.append(
            "[VERIFIED DATA]"
        )

        output.append(
            "The following values come directly from the "
            "NSE scanner output."
        )

        output.append(
            "[CALCULATED DATA]"
        )

        output.append(
            "The scanner calculates technical scores, returns, "
            "breadth, momentum-related rankings, relative strength, "
            "volume metrics and trend classifications."
        )

        output.append(
            "[INFERENCE BOUNDARY]"
        )

        output.append(
            "The LLM may interpret verified/calculated values, "
            "but must clearly distinguish interpretation from measurement."
        )

        output.append(
            "[NOT AVAILABLE]"
        )

        output.append(
            "The current analysis layer does not provide verified "
            "India VIX, sector breadth, derivatives/OI, institutional "
            "flow, event-risk analysis, transaction-cost modeling, "
            "or portfolio covariance analysis."
        )

        output.append(
            "These unavailable fields MUST NOT be inferred or invented."
        )

        # MARKET REGIME
        # ====================================================================

        output.append("")
        output.append(
            "==================== MARKET REGIME ======================="
        )

            f"[CALCULATED DATA] MARKET_REGIME={market_regime}"
        )

        output.append(
            "This regime is generated by the scanner's existing "
            "NIFTY-return and breadth methodology."
        )

        # NIFTY BENCHMARK
        # ====================================================================

        output.append("")
        output.append(
            "==================== NIFTY BENCHMARK ======================"
        )

            f"{_fmt_percent(nifty_returns.get('return_5d'))}"
        )

        output.append(
            f"NIFTY_RETURN_20D="
            f"{_fmt_percent(nifty_returns.get('return_20d'))}"
        )

        output.append(
            f"NIFTY_RETURN_60D="
            f"{_fmt_percent(nifty_returns.get('return_60d'))}"
        )

        # MARKET BREADTH
